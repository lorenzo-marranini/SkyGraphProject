LOAD CSV WITH HEADERS FROM 'file:///cities.csv' AS row
MERGE (c:City {city_state: row.city_state})
SET c.name = row.city,
    c.state_id = row.state_id,
    c.population = toInteger(row.population),
    c.density = toInteger(row.density),
    c.timezone = row.timezone;





LOAD CSV WITH HEADERS FROM 'file:///airports.csv' AS row
// Create the Airport node
MERGE (a:Airport {iata_code: row.iata_code})
SET a.name = row.airport,
    a.latitude = toFloat(row.latitude),
    a.longitude = toFloat(row.longitude)

// Find the matching City and create the relationship
WITH a, row
MATCH (c:City {city_state: row.city_state})
MERGE (a)-[:LOCATED_IN]->(c);



LOAD CSV WITH HEADERS FROM 'file:///routes.csv' AS row

// Filtro di sicurezza per evitare righe vuote
WITH row WHERE row.ORIGIN_AIRPORT IS NOT NULL AND row.DESTINATION_AIRPORT IS NOT NULL

MATCH (source:Airport {iata_code: row.ORIGIN_AIRPORT})
MATCH (target:Airport {iata_code: row.DESTINATION_AIRPORT})

MERGE (source)-[r:ROUTE]->(target)
SET 
    r.num_voli = toInteger(row.num_voli),
    r.mean_scheduled_time = toFloat(row.mean_scheduled_time);
